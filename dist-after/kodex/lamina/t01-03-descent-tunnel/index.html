03. Polar / Radial Distortion (Pseudo Code) 
// polar tunnel mapping
vec2 polarDistort(vec2 uv, float depth){
   vec2 p = uv * 2.0 - 1.0;
   float r  = length(p);
   float a  = atan(p.y, p.x);
   float k  = 1.0/(1.0 + r * depth);
   float rd = pow(r, k);
   vec2 q = vec2(cos(a), sin(a)) * rd;
   return q * 0.5 + 0.5;
}
